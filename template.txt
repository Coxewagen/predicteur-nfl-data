  function weekState(D2, selected) {
    const avail = D2.weeksAvailable && D2.weeksAvailable.length ? D2.weeksAvailable : [D2.week];
    const minWeek = Math.min(...avail), maxWeek = Math.max(...avail);
    let w = selected === null || selected === void 0 ? D2.week : selected;
    w = Math.max(minWeek, Math.min(maxWeek, w));
    const title = w === D2.week ? "Matchs de la semaine " + w : w < D2.week ? "Semaine " + w + " (termin\xE9e)" : "Semaine " + w + " (\xE0 venir)";
    const games = D2.weeks && D2.weeks[String(w)] || (w === D2.week ? D2.games : []) || [];
    return { w, minWeek, maxWeek, title, games, future: w > D2.week };
  }
  function gameCard(D2, ctx, g, idx) {
    const p = g.pred, done = g.status === "closed";
    const favHome = p ? p.pHome >= p.pAway : null;
    const top = p ? Math.max(p.pHome, p.pAway) : 0;
    const unlocked = ctx.isUnlocked(g);
    const net = p && unlocked && top >= D2.safe;
    let tag = "";
    if (done && p && unlocked) {
      const ok = g.homeScore > g.awayScore === favHome;
      tag = g.homeScore === g.awayScore ? '<span class="tag">Match nul</span>' : '<span class="tag ' + (ok ? "ok" : "ko") + '">' + ic(ok ? "check" : "x") + (ok ? "Pr\xE9vision juste" : "Pr\xE9vision fausse") + "</span>";
    } else if (done) tag = '<span class="tag">Termin\xE9</span>';
    else if (net) tag = '<span class="tag net">' + ic("star") + "Favori net</span>";
    const chips = (a) => (a.qb ? '<span class="chip qb" title="Quarterback absent">' + ic("cross") + "QB</span>" : "") + (a.off + a.deff ? '<span class="chip" title="Autres titulaires absents">' + ic("cross") + (a.off + a.deff) + "</span>" : "");
    const line = (side) => {
      const name = g[side];
      const showPct = unlocked && p;
      const prob = side === "home" ? p && p.pHome : p && p.pAway;
      const fav = showPct && (side === "home" ? favHome === true : favHome === false);
      const badge = showPct ? '<span class="pbadge ' + side + (fav ? " fav" : "") + '">' + pct(prob) + "</span>" : "";
      const right = done ? badge + '<span class="pct' + (fav ? " fav" : "") + '">' + (side === "home" ? g.homeScore : g.awayScore) + "</span>" : showPct ? badge : '<span class="pct">\u2013</span>';
      return '<div class="team">' + shield(D2, name) + '<span class="tname' + (fav ? " fav" : "") + '"><span class="long">' + esc(name) + '</span><span class="short">' + esc(nick(D2, name)) + '</span><span class="role" title="' + (side === "home" ? "\xC0 domicile" : "\xC0 l\u2019ext\xE9rieur") + '">' + ic(side === "home" ? "home" : "plane") + "</span>" + chips(g.abs[side]) + '</span><span class="pright">' + right + "</span></div>";
    };
    const foot = [];
    if (done) foot.push("R\xE9sultat : <b>" + g.homeScore + " \u2013 " + g.awayScore + "</b>");
    if (p && unlocked) foot.push("Score attendu : <b>" + p.expHome + " \u2013 " + p.expAway + "</b>");
    if (!p && !done) foot.push("Pr\xE9vision indisponible (pas encore assez de matchs jou\xE9s)");
    const lock = p && !unlocked ? ctx.lockRow(g) : "";
    return '<div class="game' + (net && !done ? " net" : "") + (isBet(g.id) ? " betted" : "") + ' clickable" data-idx="' + idx + '" data-id="' + esc(g.id) + '"><div class="ghead"><span class="when">' + ic("clock") + when(g.kickoff) + '</span><span class="gtags">' + validated() + tag + "</span></div>" + line("home") + line("away") + lock + (foot.length ? '<div class="gfoot">' + foot.map((f) => "<span>" + f + "</span>").join("") + "</div>" : "") + betRow(g) + "</div>";
  }
  function absBlockHtml(D2, name) {
    const items = ((D2.absences || []).find((a) => a.team === name) || { items: [] }).items;
    if (!items.length) return '<p class="meta">Aucune absence notable retenue.</p>';
    return '<ul class="abs">' + items.map((i) => "<li>" + esc(i.side) + " : " + esc(i.txt) + "</li>").join("") + "</ul>";
  }
  var formHtml = (D2, t) => {
    const f = (D2.fiche || {})[t];
    if (!f || !f.form || !f.form.length) return "\u2013";
    return f.form.map((x) => '<span class="' + (x.r === "V" ? "yes" : x.r === "D" ? "no" : "") + '"><b>' + x.r + "</b></span> " + esc(x.s) + ' <span class="dim">vs ' + esc((D2.teams[x.opp] || {}).abbr || "?") + "</span>").join("<br>");
  };
  function modalHtml(D2, ctx, g) {
    var _a, _b, _c, _d;
    const p = g.pred, done = g.status === "closed";
    const unlocked = ctx.isUnlocked(g);
    const fh = (D2.fiche || {})[g.home] || {}, fa = (D2.fiche || {})[g.away] || {};
    let pred;
    if (unlocked && p) {
      const favHome = p.pHome >= p.pAway, ecart = Math.abs(p.expHome - p.expAway);
      const verdict = done && g.homeScore !== g.awayScore ? '<p class="verdict ' + (g.homeScore > g.awayScore === favHome ? "yes" : "no") + '">' + ic(g.homeScore > g.awayScore === favHome ? "check" : "x") + (g.homeScore > g.awayScore === favHome ? "Pr\xE9vision juste" : "Pr\xE9vision fausse") + " \u2014 r\xE9sultat " + g.homeScore + " \u2013 " + g.awayScore + " (pr\xE9vu " + p.expHome + " \u2013 " + p.expAway + ")</p>" : "";
      pred = '<div class="bigbadges"><span class="pbadge big home' + (favHome ? " fav" : "") + '"><i>Domicile</i>' + pct(p.pHome) + '</span><span class="pbadge big away' + (!favHome ? " fav" : "") + '"><i>Ext\xE9rieur</i>' + pct(p.pAway) + '</span></div><div class="gfoot"><span>Score attendu : <b>' + p.expHome + " \u2013 " + p.expAway + "</b></span><span>\xC9cart attendu : <b>" + (ecart ? esc(nick(D2, favHome ? g.home : g.away)) + " de " + ecart + " pt" + (ecart > 1 ? "s" : "") : "aucun") + "</b></span></div>" + verdict + (g.prov ? '<p class="meta">Pr\xE9vision provisoire : recalcul\xE9e \xE0 chaque mise \xE0 jour avec les stats et les absences du moment.</p>' : "");
    } else if (p) {
      pred = ctx.gateCard();
    } else {
      pred = '<p class="meta">Pas assez de matchs jou\xE9s pour g\xE9n\xE9rer une pr\xE9vision.</p>';
    }
    const stats = '<table class="cmp"><thead><tr><th></th><th>' + esc(nick(D2, g.home)) + "</th><th>" + esc(nick(D2, g.away)) + "</th></tr></thead><tbody><tr><td>Bilan</td><td>" + esc(fh.rec || "\u2013") + "</td><td>" + esc(fa.rec || "\u2013") + "</td></tr><tr><td>Points marqu\xE9s / match</td><td>" + ((_a = fh.pf) != null ? _a : "\u2013") + "</td><td>" + ((_b = fa.pf) != null ? _b : "\u2013") + "</td></tr><tr><td>Points encaiss\xE9s / match</td><td>" + ((_c = fh.pa) != null ? _c : "\u2013") + "</td><td>" + ((_d = fa.pa) != null ? _d : "\u2013") + "</td></tr><tr><td>3 derniers matchs</td><td>" + formHtml(D2, g.home) + "</td><td>" + formHtml(D2, g.away) + "</td></tr></tbody></table>";
    const abs = unlocked && p ? '<h3>Absences retenues</h3><div class="absgrid"><div><b>' + esc(nick(D2, g.home)) + "</b>" + absBlockHtml(D2, g.home) + "</div><div><b>" + esc(nick(D2, g.away)) + "</b>" + absBlockHtml(D2, g.away) + "</div></div>" : "";
    const notes = [];
    if (g.neutral) notes.push("Match jou\xE9 \xE0 l\u2019\xE9tranger : pas d\u2019avantage du terrain.");
    return '<div class="handle"></div><button class="modal-close" id="modalClose" aria-label="Fermer">\u2715</button><div class="mwhen' + (isBet(g.id) ? " betted" : "") + '" data-id="' + esc(g.id) + '"><span class="when">' + ic("clock") + when(g.kickoff) + "</span>" + validated() + '</div><div class="team">' + shield(D2, g.home) + '<span class="tname"><span class="long">' + esc(g.home) + '</span><span class="role">' + ic("home") + '</span></span></div><div class="team">' + shield(D2, g.away) + '<span class="tname"><span class="long">' + esc(g.away) + '</span><span class="role">' + ic("plane") + "</span></span></div>" + betRow(g) + pred + stats + abs + (notes.length ? '<p class="meta">' + notes.join(" ") + "</p>" : "");
  }
  function wireGames(gamesEl, getGames, open) {
    gamesEl.addEventListener("change", (e) => {
      const i = e.target.closest("input[data-bet]");
      if (i) setBet(i.dataset.bet, i.checked);
    });
    gamesEl.addEventListener("click", (e) => {
      if (e.target.closest(".bet") || e.target.closest("[data-go-pro]")) return;
      const card = e.target.closest(".game");
      if (!card) return;
      const g = getGames()[+card.dataset.idx];
      if (g) open(g);
    });
  }
  function wirePanel(panelEl) {
    panelEl.addEventListener("change", (e) => {
      const i = e.target.closest("input[data-bet]");
      if (i) setBet(i.dataset.bet, i.checked);
    });
  }

  // src/site.js
  var $ = (id) => document.getElementById(id);
  var CTX = { isUnlocked: () => true, lockRow: () => "", gateCard: () => "" };
  var selectedWeek = null;
  var shown = [];
  function closeModal() {
    $("gameModalOverlay").classList.remove("open");
  }
  function openModal(g) {
    $("gameModalPanel").innerHTML = modalHtml(D, CTX, g);
    $("modalClose").addEventListener("click", closeModal);
    $("gameModalOverlay").classList.add("open");
  }
  function render() {
    $("eyebrow").textContent = "Saison " + D.season + " \xB7 Semaine " + D.week;
    const ws = weekState(D, selectedWeek);
    selectedWeek = ws.w;
    shown = ws.games;
    $("weekTitle").textContent = ws.title;
    $("updatedMeta").textContent = "Mis \xE0 jour le " + D.updatedLabel + " (heure de Paris)" + (ws.future ? " \xB7 pr\xE9visions provisoires" : "");
    $("weekPrev").disabled = selectedWeek <= ws.minWeek;
    $("weekNext").disabled = selectedWeek >= ws.maxWeek;
    $("games").innerHTML = shown.length ? shown.map((g, i) => gameCard(D, CTX, g, i)).join("") : '<p class="meta">Aucun match enregistr\xE9 pour cette semaine.</p>';
  }
  $("weekPrev").addEventListener("click", () => {
    if (!$("weekPrev").disabled) {
      selectedWeek--;
      render();
    }
  });
  $("weekNext").addEventListener("click", () => {
    if (!$("weekNext").disabled) {
      selectedWeek++;
      render();
    }
  });
  wireGames($("games"), () => shown, openModal);
  wirePanel($("gameModalPanel"));
  $("gameModalOverlay").addEventListener("click", (e) => {
    if (e.target.id === "gameModalOverlay") closeModal();
  });
  document.addEventListener("keydown", (e) => {
    if (e.key === "Escape") closeModal();
  });
  render();
})();
