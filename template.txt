<script>
const D = __DATA__;
const $ = id => document.getElementById(id);
const esc = s => String(s).replace(/[&<>"]/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;'}[c]));
const pct = p => Math.round(p*100) + ' %';
const ic = n => '<svg class="ic" aria-hidden="true"><use href="#i-' + n + '"/></svg>';
const P = {timeZone:'Europe/Paris'};
function when(iso){
  const d = new Date(iso);
  return d.toLocaleDateString('fr-FR', {...P, weekday:'short', day:'numeric', month:'short'}) + ' · ' +
         d.toLocaleTimeString('fr-FR', {...P, hour:'2-digit', minute:'2-digit'});
}
// Blason d'illustration : écu simple aux deux couleurs de l'équipe, avec son sigle.
function lum(hex){ const n = parseInt(hex.slice(1), 16); return (0.299*(n>>16) + 0.587*((n>>8)&255) + 0.114*(n&255)) / 255; }
function shield(name){
  const t = D.teams[name] || {abbr:'?', c1:'#445566', c2:'#FFFFFF'};
  const txt = lum(t.c1) > 0.6 ? '#0D1B2A' : '#FFFFFF';
  const fs = t.abbr.length > 2 ? 12.5 : 15;
  return '<svg class="shield" viewBox="0 0 38 42" role="img" aria-label="' + esc(name) + '">' +
    '<path d="M3 3h32v19c0 9-7 14.5-16 17.5C10 36.5 3 31 3 22z" fill="none" stroke="#8795A8" stroke-width="4.6" stroke-linejoin="round"/>' +
    '<path d="M3 3h32v19c0 9-7 14.5-16 17.5C10 36.5 3 31 3 22z" fill="' + t.c1 + '" stroke="' + t.c2 + '" stroke-width="3" stroke-linejoin="round"/>' +
    '<text x="19" y="23" text-anchor="middle" font-family="Arial Narrow,Arial,sans-serif" font-weight="700" font-size="' + fs + '" fill="' + txt + '">' + esc(t.abbr) + '</text></svg>';
}
const nick = name => (D.teams[name] && D.teams[name].nick) || name;

$('eyebrow').textContent = 'Saison ' + D.season + ' · Semaine ' + D.week;
$('meta').textContent = 'Mis à jour le ' + D.updatedLabel + ' (heure de Paris)';
$('weekTitle').textContent = 'Matchs de la semaine ' + D.week;

const R = D.rapport, nCtl = R.nouveaux.length + R.revenus.length + R.qb.length;
if(nCtl){
  const nNew = R.nouveaux.length + R.qb.length, pl = (n, a, b) => n + ' ' + (n > 1 ? b : a);
  $('alert').hidden = false;
  $('alert').innerHTML = ic('warn') + '<span><b>Contrôle des absences :</b> ' +
    (nNew ? pl(nNew, 'titulaire indisponible selon les données officielles n\'est pas pris', 'titulaires indisponibles selon les données officielles ne sont pas pris') + ' en compte dans le calcul' : '') +
    (nNew && R.revenus.length ? ', et ' : '') +
    (R.revenus.length ? pl(R.revenus.length, 'absence retenue semble terminée', 'absences retenues semblent terminées') : '') +
    '. Probabilités à lire avec prudence. Détail en bas de page.</span>';
}

$('games').innerHTML = D.games.map(g => {
  const p = g.pred, done = g.status === 'closed';
  const favHome = p ? p.pHome >= p.pAway : null, top = p ? Math.max(p.pHome, p.pAway) : 0;
  const net = p && top >= D.safe;
  let tag = '';
  if(done && p){
    const ok = (g.homeScore > g.awayScore) === favHome;
    tag = g.homeScore === g.awayScore ? '<span class="tag">Match nul</span>'
        : '<span class="tag ' + (ok?'ok':'ko') + '">' + ic(ok?'check':'x') + (ok?'Prévision juste':'Prévision fausse') + '</span>';
  } else if(done) tag = '<span class="tag">Terminé</span>';
  else if(net) tag = '<span class="tag net">' + ic('star') + 'Favori net</span>';
  const chips = a => (a.qb ? '<span class="chip qb" title="Quarterback absent">' + ic('cross') + 'QB</span>' : '') +
    (a.off + a.deff ? '<span class="chip" title="Autres titulaires absents">' + ic('cross') + (a.off + a.deff) + '</span>' : '');
  const line = (side, prob, fav) => { const name = g[side];
    return '<div class="team">' + shield(name) + '<span class="tname' + (fav?' fav':'') + '"><span class="long">' + esc(name) + '</span><span class="short">' + esc(nick(name)) +
      '</span><span class="role" title="' + (side==='home'?'À domicile':'À l\'extérieur') + '">' + ic(side==='home'?'home':'plane') + '</span>' + chips(g.abs[side]) + '</span>' +
      '<span class="pct' + (fav?' fav':'') + '">' + (done && !p ? (side==='home'?g.homeScore:g.awayScore) : prob==null ? '–' : pct(prob)) + '</span></div>'; };
  const c = s => (D.teams[g[s]] || {}).c1 || '#445566';
  const bar = p ? '<div class="split"><i style="width:' + (p.pHome*100).toFixed(1) + '%;background:' + c('home') + '"></i><i style="flex:1;background:' + c('away') + '"></i></div>' : '';
  const foot = [];
  if(done && p) foot.push('Résultat : <b>' + g.homeScore + ' – ' + g.awayScore + '</b>');
  if(p) foot.push('Score attendu : <b>' + p.expHome + ' – ' + p.expAway + '</b>');
  else foot.push('Joué avant le début du suivi : pas de prévision enregistrée');
  return '<div class="game' + (net && !done ? ' net' : '') + '"><div class="ghead"><span class="when">' + ic('clock') + when(g.kickoff) + '</span>' + tag + '</div>' +
    line('home', p && p.pHome, favHome === true) + line('away', p && p.pAway, favHome === false) + bar +
    '<div class="gfoot">' + foot.map(f => '<span>' + f + '</span>').join('') + '</div></div>';
}).join('');

const B = D.bilan, rate = (a,b) => b ? Math.round(a/b*100) + ' %' : '–';
$('figs').innerHTML =
  '<div class="fig"><span class="n">' + B.total + '</span><span class="l">prévisions vérifiées</span></div>' +
  '<div class="fig g"><span class="n">' + rate(B.correct, B.total) + '</span><span class="l">justes au total (' + B.correct + ' sur ' + B.total + ')</span></div>' +
  '<div class="fig g"><span class="n">' + rate(B.safeCorrect, B.safeTotal) + '</span><span class="l">justes en favoris nets (' + B.safeCorrect + ' sur ' + B.safeTotal + ')</span></div>';
$('bilanNote').textContent = 'Chaque prévision est figée au coup d\'envoi, puis comparée au résultat réel. Un favori net est une équipe à ' +
  Math.round(D.safe*100) + ' % ou plus.' + (B.total ? '' : ' Le suivi démarre avec la semaine ' + D.week + ' : les premiers résultats apparaîtront après les matchs.');
if(B.total){
  $('bilanTbl').innerHTML = '<table><thead><tr><th>Sem.</th><th>Match</th><th>Favori du modèle</th><th>Score</th><th>Verdict</th></tr></thead><tbody>' +
    B.list.map(j => { const fh = j.pHome >= j.pAway;
      return '<tr><td>' + j.week + '</td><td>' + esc(nick(j.home)) + ' – ' + esc(nick(j.away)) + '</td><td>' + esc(nick(fh ? j.home : j.away)) + ' (' + pct(Math.max(j.pHome, j.pAway)) +
        ')</td><td>' + j.homeScore + ' – ' + j.awayScore + '</td><td class="' + (j.correct?'yes':'no') + '">' + (j.correct?'Juste':'Faux') + '</td></tr>'; }).join('') +
    '</tbody></table>';
}

const nAbs = D.absences.reduce((s,t) => s + t.items.length, 0);
$('absSum').textContent = 'Absences prises en compte dans le calcul (' + nAbs + ')';
$('absList').innerHTML = D.absences.map(t => '<div><h3>' + shield(t.team) + esc(t.team) + '</h3><ul>' +
  t.items.map(i => '<li>' + i.side + ' : ' + esc(i.txt) + (i.tier ? ' · niveau ' + esc(i.tier) : '') + '</li>').join('') + '</ul></div>').join('');
$('ctlSum').textContent = 'Absences à vérifier (' + nCtl + ')';
const block = (title, arr) => arr.length ? '<div><h3>' + title + '</h3><ul>' + arr.map(x => '<li>' + esc(x) + '</li>').join('') + '</ul></div>' : '';
$('ctlList').innerHTML = (block('Quarterbacks', R.qb) + block('Titulaires indisponibles non pris en compte', R.nouveaux) +
  block('Absences retenues qui semblent terminées', R.revenus)) || '<p style="margin:0">Aucun écart entre les absences retenues et les données officielles.</p>';
</script>
